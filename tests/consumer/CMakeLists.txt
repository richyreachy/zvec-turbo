cmake_minimum_required(VERSION 3.18)
project(turbo_consumer LANGUAGES CXX)
if(TURBO_SOURCE_DIR)
  add_subdirectory("${TURBO_SOURCE_DIR}" turbo)
else()
  find_package(Turbo 0.1 CONFIG REQUIRED)
endif()
add_executable(consumer main.cc)
target_link_libraries(consumer PRIVATE turbo::turbo)
enable_testing()
add_test(NAME consumer COMMAND consumer)
