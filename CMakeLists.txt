cmake_minimum_required(VERSION 3.18)
project(Turbo VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CheckCXXSourceCompiles)

set(TURBO_TOP_LEVEL OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(TURBO_TOP_LEVEL ON)
endif()
option(TURBO_BUILD_TESTS "Build standalone tests (no downloaded dependencies)" ${TURBO_TOP_LEVEL})
option(TURBO_BUILD_EXAMPLES "Build example and target inspector" ${TURBO_TOP_LEVEL})
option(TURBO_ENABLE_INSTALL "Generate install rules" ${TURBO_TOP_LEVEL})
option(TURBO_ENABLE_SIMD "Enable runtime SIMD selection" ON)
option(TURBO_ENABLE_AVX512 "Build AVX512 variants when supported" ON)
option(TURBO_BUILD_SHARED "Build a shared library" OFF)
option(TURBO_ENABLE_SANITIZERS "Enable address/undefined sanitizers (GCC/Clang)" OFF)
set(TURBO_SANITIZERS "address,undefined" CACHE STRING "Sanitizers used when TURBO_ENABLE_SANITIZERS=ON")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Turbo 0.1 supports 64-bit targets only (FHT context layout).")
endif()
list(LENGTH CMAKE_OSX_ARCHITECTURES _turbo_arch_count)
if(_turbo_arch_count GREATER 1)
  message(FATAL_ERROR "Build one Apple architecture per build directory; universal binaries are not supported yet.")
endif()
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" TURBO_PROCESSOR)
if(APPLE AND CMAKE_OSX_ARCHITECTURES)
  string(TOLOWER "${CMAKE_OSX_ARCHITECTURES}" TURBO_PROCESSOR)
endif()

# The baseline/dispatcher never inherits our optional ISA flags. Keep variant
# flags per translation unit and turn off CMake IPO across those boundaries.
file(GLOB_RECURSE TURBO_SOURCES CONFIGURE_DEPENDS src/*.cc)
if(TURBO_BUILD_SHARED)
  add_library(turbo SHARED ${TURBO_SOURCES})
  target_compile_definitions(turbo PUBLIC TURBO_SHARED PRIVATE TURBO_BUILDING_LIBRARY)
else()
  add_library(turbo STATIC ${TURBO_SOURCES})
endif()
add_library(turbo::turbo ALIAS turbo)
target_compile_features(turbo PUBLIC cxx_std_17)
target_include_directories(turbo PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
  PRIVATE src src/distance "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_target_properties(turbo PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
  INTERPROCEDURAL_OPTIMIZATION FALSE VERSION ${PROJECT_VERSION} SOVERSION 0.1)

foreach(isa SSE2 AVX2 AVX512 VNNI FP16 NEON)
  set(TURBO_BUILD_${isa} 0)
endforeach()
set(TURBO_HAVE_ARM_FP16_KERNEL 0)

function(turbo_variant name flags probe)
  set(CMAKE_REQUIRED_FLAGS "${flags}")
  check_cxx_source_compiles("${probe}" TURBO_COMPILER_${name})
  if(TURBO_COMPILER_${name})
    separate_arguments(_flags NATIVE_COMMAND "${flags}")
    set_source_files_properties(${ARGN} PROPERTIES COMPILE_OPTIONS "${_flags}")
    set(TURBO_BUILD_${name} 1 PARENT_SCOPE)
  endif()
endfunction()

if(TURBO_PROCESSOR MATCHES "^(x86_64|amd64|x64)$")
  if(NOT MSVC)
    target_compile_options(turbo PRIVATE -march=x86-64 -mno-avx -mno-avx2 -mno-avx512f -mno-fma -mno-f16c)
  endif()
  if(TURBO_ENABLE_SIMD)
    set(TURBO_BUILD_SSE2 1)
    file(GLOB_RECURSE avx2_sources src/distance/avx2/*.cc)
    if(MSVC)
      set(avx2_flags "/arch:AVX2")
    else()
      set(avx2_flags "-mavx2 -mfma -mf16c")
    endif()
    turbo_variant(AVX2 "${avx2_flags}" "#include <immintrin.h>\n#ifndef __AVX2__\n#error AVX2 missing\n#endif\nint main(){return 0;}" ${avx2_sources})
    # MSVC AVX512 has a broader compiler-defined ISA bundle. Keep it disabled
    # until that bundle is verified on Windows hardware; AVX2 stays available.
    if(TURBO_ENABLE_AVX512 AND NOT MSVC)
      set(avx512_flags "${avx2_flags} -mavx512f -mavx512bw -mavx512dq -mavx512vl")
      file(GLOB_RECURSE avx512_sources src/distance/avx512/*.cc src/conversion/avx512/*.cc)
      list(APPEND avx512_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/distance/avx512_vnni/fp16/squared_euclidean.cc")
      turbo_variant(AVX512 "${avx512_flags}" "#include <immintrin.h>\nint main(){__m512 x = _mm512_setzero_ps(); return _mm512_cvtss_f32(x) != 0;}" ${avx512_sources})
      if(TURBO_BUILD_AVX512)
        file(GLOB_RECURSE vnni_sources src/distance/avx512_vnni/*.cc)
        list(REMOVE_ITEM vnni_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/distance/avx512_vnni/fp16/squared_euclidean.cc")
        turbo_variant(VNNI "${avx512_flags} -mavx512vnni" "#include <immintrin.h>\nint main(){__m512i x = _mm512_setzero_si512(); x = _mm512_dpbusd_epi32(x,x,x); return _mm512_cvtsi512_si32(x);}" ${vnni_sources})
        file(GLOB_RECURSE fp16_sources src/distance/avx512_fp16/*.cc)
        turbo_variant(FP16 "${avx512_flags} -mavx512fp16" "#include <immintrin.h>\nint main(){__m512h x = _mm512_setzero_ph(); x = _mm512_add_ph(x,x); return _mm512_cvtsi512_si32(_mm512_castph_si512(x));}" ${fp16_sources})
      endif()
    endif()
  endif()
elseif(TURBO_PROCESSOR MATCHES "^(aarch64|arm64)$")
  if(NOT MSVC)
    target_compile_options(turbo PRIVATE -march=armv8-a)
  endif()
  if(TURBO_ENABLE_SIMD)
    set(TURBO_BUILD_NEON 1)
    if(NOT MSVC)
      file(GLOB_RECURSE neon_fp16_sources src/distance/neon_fp16/*.cc)
      turbo_variant(NEON_FP16 "-march=armv8.2-a+fp16" "#include <arm_neon.h>\nint main(){float16x8_t x=vdupq_n_f16(1); x=vfmaq_f16(x,x,x); return vgetq_lane_u16(vreinterpretq_u16_f16(x),0)==0;}" ${neon_fp16_sources})
      if(TURBO_BUILD_NEON_FP16)
        set(TURBO_HAVE_ARM_FP16_KERNEL 1)
        target_compile_definitions(turbo PRIVATE TURBO_HAVE_ARM_FP16_KERNEL=1)
      endif()
    endif()
  endif()
endif()
configure_file(cmake/turbo_config.h.in generated/turbo_config.h @ONLY)
message(STATUS "Turbo variants: SSE2=${TURBO_BUILD_SSE2} AVX2=${TURBO_BUILD_AVX2} AVX512=${TURBO_BUILD_AVX512} VNNI=${TURBO_BUILD_VNNI} FP16=${TURBO_BUILD_FP16} NEON=${TURBO_BUILD_NEON}")

  target_link_options(turbo PUBLIC "-fsanitize=${TURBO_SANITIZERS}")

  configure_file(cmake/turbo.pc.in turbo.pc @ONLY)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/turbo.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
